-- ===== process_x_glossary =====

CREATE TABLE `process_x_glossary` (
  `id` int(11) NOT NULL AUTO_INCREMENT,
  `processid` int(11) DEFAULT NULL,
  `glossaryid` int(11) DEFAULT NULL,
  `relationtype` int(11) DEFAULT NULL,
  `description` text DEFAULT NULL,
  `CreateDatetime` datetime DEFAULT NULL,
  `LastUpdateDatetime` datetime DEFAULT NULL,
  `LastUpdate_UserID` int(11) DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `fk_pxg_process_idx` (`processid`),
  KEY `fk_pxg_glossary_idx` (`glossaryid`),
  KEY `fk_pxg_reltype_idx` (`relationtype`),
  KEY `fk_pxg_user_idx` (`LastUpdate_UserID`),
  CONSTRAINT `fk_pxg_process` FOREIGN KEY (`processid`) REFERENCES `process` (`id`),
  CONSTRAINT `fk_pxg_glossary` FOREIGN KEY (`glossaryid`) REFERENCES `glossary` (`ID`),
  CONSTRAINT `fk_pxg_reltype` FOREIGN KEY (`relationtype`) REFERENCES `process_x_glossary_relationtype` (`id`),
  CONSTRAINT `fk_pxg_user` FOREIGN KEY (`LastUpdate_UserID`) REFERENCES `people` (`ID`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_general_ci;

القسم 8: ملاحظات إضافية
================================================================================

1. تباين في naming conventions:
   - Policy tables تستخدم: Policy_ID, Glossary_ID (PascalCase + Underscore)
   - Process tables تستخدم: processid, datasetid (camelCase)
   - Project tables تستخدم: projectid, systemid (camelCase)
   
2. جميع الجداول تتبع نفس البنية الأساسية:
   - ID/id (Primary Key)
   - [SourceObject]_id (Foreign Key للكائن الأول)
   - [TargetObject]_id (Foreign Key للكائن الثاني)
   - relationtype/Relation_Type (Foreign Key لنوع العلاقة)
   - description/Description (نص وصفي)
   - CreateDatetime (تاريخ الإنشاء)
   - LastUpdateDatetime (تاريخ آخر تحديث)
   - LastUpdate_UserID (المستخدم الذي قام بآخر تحديث)

3. كل جدول relationship له جدول relationtype مصاحب يحتوي على:
   - PrimaryName (اسم العلاقة)
   - Description (وصف العلاقة)
   - Priority (الأولوية)
   - ReverseName (اسم العلاقة العكسية)

private List<Map<String, Object>> loadSystemRows(Set<Integer> systemIds, String correlationId) {
    if (systemIds == null || systemIds.isEmpty()) return new ArrayList<>();
    
    // Verify correct ID column and table alias (NEW)
    String tableName = "system";
    String idColumn = "id"; // NOT "ID" - use lowercase for system table
    String tableAlias = "s";
    
    try {
        DatabaseHelper dbHelper = new DatabaseHelper();
        String placeholders = String.join(",", Collections.nCopies(systemIds.size(), "?"));
        
        // Use correct alias.column format (NEW)
        String sql = "SELECT " + tableAlias + "." + idColumn + ", " + tableAlias + ".Name, " + tableAlias + ".Long_Name, " + 
                     tableAlias + ".Description, " + tableAlias + ".AssetID AS Ref_Number, " + 
                     tableAlias + ".parent_id AS Parent_ID, " + tableAlias + ".Deleted_datetime AS DeletedDatetime " +
                     "FROM " + tableName + " " + tableAlias + 
                     " WHERE " + tableAlias + "." + idColumn + " IN (" + placeholders + ") " +
                     "AND " + tableAlias + ".Deleted_datetime IS NULL";
        
        List<Object> params = new ArrayList<>(systemIds);
        List<Map<String, Object>> results = dbHelper.executeQuery(sql, params);
        
        // Log warning if mismatch (NEW)
        if (results.size() != systemIds.size()) {
            System.err.println("[UnisonSearchService] WARNING: SuperAdmin missing IDs after fetch for facet SYSTEM: " +
                    "expected " + systemIds.size() + " got " + results.size() + 
                    " missing " + getMissingIds(systemIds, results, idColumn) +
                    " sqlHash=" + sql.hashCode() + " module=system idColumn=" + tableAlias + "." + idColumn);
        }
        
        return results;
    } catch (Exception e) {
        System.err.println("[UnisonSearchService] Error in loadSystemRows: " + e.getMessage());
        return new ArrayList<>();
    }
}

// Helper method (NEW):
private Set<Integer> getMissingIds(Set<Integer> expected, List<Map<String, Object>> results, String idColumn) {
    Set<Integer> returned = new HashSet<>();
    for (Map<String, Object> row : results) {
        Object idObj = row.get(idColumn);
        if (idObj == null) {
            idObj = row.get(idColumn.toUpperCase());
        }
        if (idObj == null) {
            idObj = row.get("id");
        }
        if (idObj == null) {
            idObj = row.get("ID");
        }
        if (idObj != null) {
            returned.add(Integer.valueOf(idObj.toString()));
        }
    }
    Set<Integer> missing = new HashSet<>(expected);
    missing.removeAll(returned);
    return missing;
}
